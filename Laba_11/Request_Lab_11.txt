CREATE INDEX idx_contract_fk ON contract(id_client, id_realtor, id_realty);

EXPLAIN ANALYZE
SELECT client.second_name, CONCAT(r.street, ' ', r.house) AS adress
FROM contract
JOIN client ON client.id = contract.id_client
JOIN realty r ON r.id = contract.id_realty
WHERE id_client <= 5 AND id_realtor > 5;

EXPLAIN ANALYZE
SELECT COUNT(*) FROM contract
WHERE id_client <= 5 AND id_realtor > 5;

DROP INDEX idx_contract_fk;

//Составной индекс на 2 строки и особенности

CREATE INDEX idx_perf_test_reason_annotation ON perf_test(reason, annotation);

EXPLAIN ANALYZE 
SELECT * FROM perf_test
WHERE reason LIKE('a%') AND annotation LIKE('AB%'); //Сканирование по индексу

EXPLAIN ANALYZE 
SELECT * FROM perf_test
WHERE reason LIKE('abc%'); //Сканирование по индексу

EXPLAIN ANALYZE 
SELECT * FROM perf_test
WHERE annotation LIKE('ABC%'); //Последовательное сканирование
