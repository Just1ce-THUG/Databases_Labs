CREATE OR REPLACE FUNCTION delete_client_cascade()
RETURNS TRIGGER
LANGUAGE plpgsql
AS $$
BEGIN
	DELETE FROM contract
	WHERE id_client = OLD.id;
	
	RETURN OLD;
END;
$$;

CREATE TRIGGER t_delete_client_cascade
BEFORE DELETE ON client
FOR EACH ROW
WHEN (OLD.id % 2 = 0)
EXECUTE FUNCTION delete_client_cascade();

BEGIN;
DELETE FROM client WHERE id = 6;

SELECT * FROM client WHERE id = 6;

ROLLBACK;

CREATE SEQUENCE sq_client_id START 10 INCREMENT 5;

CREATE OR REPLACE FUNCTION set_id_client()
RETURNS TRIGGER
LANGUAGE plpgsql
AS $$
BEGIN
	IF NEW.id IS NULL THEN
		NEW.id := nextval('sq_client_id');
	END IF;

	IF NEW.e_mail IS NULL THEN
		RAISE NOTICE 'The client with id: % without e-mail', NEW.id;
	END IF;

	RETURN NEW;
END;
$$;

CREATE SEQUENCE sq_log_client_id;

CREATE TABLE log_client (
	id INT PRIMARY KEY DEFAULT nextval('sq_log_client_id'),
	log_time TIMESTAMP DEFAULT now(),
	user_name TEXT,
	dml_type VARCHAR(20),
	client_id_old INT,
	client_id_new INT,
	client_first_name_old VARCHAR(32),
	client_first_name_new VARCHAR(32),
	client_second_name_old VARCHAR(32),
	client_second_name_new VARCHAR(32)
);

CREATE OR REPLACE FUNCTION log_client_change()
RETURNS TRIGGER
LANGUAGE plpgsql
AS $$
BEGIN
	CASE TG_OP
		WHEN 'INSERT' THEN
			INSERT INTO log_client (user_name, dml_type, 
									client_id_new, client_first_name_new, 
									client_second_name_new) VALUES
			(current_user, 'INSERT', NEW.id, NEW.first_name, NEW.second_name);
			
			RETURN NEW;
		WHEN 'UPDATE' THEN
			INSERT INTO log_client (user_name, dml_type, client_id_old, client_id_new,
									client_first_name_old, client_first_name_new,
									client_second_name_old, client_second_name_new) VALUES
			(current_user, 'UPDATE', OLD.id, NEW.id, OLD.first_name, NEW.first_name, OLD.second_name, NEW.second_name);
			
			RETURN NEW;
		WHEN 'DELETE' THEN
			INSERT INTO log_client (user_name, dml_type, 
									client_id_old, client_first_name_old, 
									client_second_name_old) VALUES
			(current_user, 'DELETE', OLD.id, OLD.first_name, OLD.second_name);
			
			RETURN OLD;
	END CASE;
END;
$$;

CREATE TRIGGER t_log_client_change
AFTER INSERT OR UPDATE OR DELETE ON client
FOR EACH ROW
EXECUTE FUNCTION log_client_change();

CREATE OR REPLACE FUNCTION insert_warning_client_second_name()
RETURNS TRIGGER
LANGUAGE plpgsql
AS $$
DECLARE
	count_same_second_name INT;
BEGIN
	SELECT COUNT(*) INTO count_same_second_name
	FROM client
	WHERE second_name = NEW.second_name;

	IF count_same_second_name != 0 THEN
		RAISE NOTICE 'The client with second_name: % is already exist', NEW.second_name;
		RETURN NULL;
	ELSE
		RETURN NEW;
	END IF;
END;
$$;

CREATE OR REPLACE FUNCTION set_total_price_contract() //Расчет общей стоимости недвижимости (цена_недвижки * кол-во_дней)
RETURNS TRIGGER
LANGUAGE plpgsql
AS $$
DECLARE
	days_count BIGINT;
	apartment_price BIGINT;
BEGIN

	IF NEW.total_price IS NULL THEN
		days_count := (NEW.date_finish::date - NEW.date_start::date)::INT;
	
		SELECT price INTO apartment_price
		FROM realty
		JOIN realty_price r_p ON r_p.id = realty.realty_price
		WHERE realty.id = NEW.id_realty;

		NEW.total_price := apartment_price * days_count;
	END IF;
	
	RETURN NEW;
END;
$$;

CREATE TRIGGER t_set_total_price_contract
BEFORE INSERT ON contract
FOR EACH ROW
EXECUTE FUNCTION set_total_price_contract();
